import pygame, math, random
pygame.init()
s = pygame.display.set_mode((800, 600))
c = pygame.time.Clock()

class Pendulum:
    def __init__(self):
        self.x, self.y = 400, 200
        self.l1, self.l2 = 150, 150
        self.m1, self.m2 = 20, 20
        self.a1, self.a2 = math.pi/2, math.pi/2
        self.v1 = self.v2 = 0
        self.g = 0.5
        self.trace = []
        
    def update(self):
        num1 = -self.g * (2 * self.m1 + self.m2) * math.sin(self.a1)
        num2 = -self.m2 * self.g * math.sin(self.a1 - 2 * self.a2)
        num3 = -2 * math.sin(self.a1 - self.a2) * self.m2
        num4 = self.v2**2 * self.l2 + self.v1**2 * self.l1 * math.cos(self.a1 - self.a2)
        denom = self.l1 * (2 * self.m1 + self.m2 - self.m2 * math.cos(2 * self.a1 - 2 * self.a2))
        
        self.v1 += (num1 + num2 + num3 * num4) / denom
        
        num1 = 2 * math.sin(self.a1 - self.a2)
        num2 = self.v1**2 * self.l1 * (self.m1 + self.m2)
        num3 = self.g * (self.m1 + self.m2) * math.cos(self.a1)
        num4 = self.v2**2 * self.l2 * self.m2 * math.cos(self.a1 - self.a2)
        denom = self.l2 * (2 * self.m1 + self.m2 - self.m2 * math.cos(2 * self.a1 - 2 * self.a2))
        
        self.v2 += (num1 * (num2 + num3 + num4)) / denom
        
        self.a1 += self.v1
        self.a2 += self.v2
        
        x1 = self.x + self.l1 * math.sin(self.a1)
        y1 = self.y + self.l1 * math.cos(self.a1)
        x2 = x1 + self.l2 * math.sin(self.a2)
        y2 = y1 + self.l2 * math.cos(self.a2)
        
        self.trace.append((x2, y2))
        if len(self.trace) > 200: self.trace.pop(0)
        
        return (x1, y1), (x2, y2)

p = Pendulum()
while True:
    for e in pygame.event.get():
        if e.type == pygame.QUIT: exit()
        if e.type == pygame.MOUSEBUTTONDOWN:
            p = Pendulum() 
    
    s.fill((0, 0, 0))
    p1, p2 = p.update()
    
    for i, pos in enumerate(p.trace):
        pygame.draw.circle(s, (255, 255, 255, i), (int(pos[0]), int(pos[1])), 1)
    
    pygame.draw.line(s, (100, 100, 100), (p.x, p.y), p1, 2)
    pygame.draw.line(s, (100, 100, 100), p1, p2, 2)
    pygame.draw.circle(s, (200, 0, 0), p1, 8)
    pygame.draw.circle(s, (0, 200, 0), p2, 8)
    
    pygame.display.flip()
    c.tick(60)
